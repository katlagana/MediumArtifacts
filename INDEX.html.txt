const starsCanvas = document.getElementById('stars');
const starsCtx = starsCanvas.getContext('2d');
let stars = [];
function resizeStars() {
  starsCanvas.width = window.innerWidth;
  starsCanvas.height = window.innerHeight;
  stars = [];
  const count = Math.floor((starsCanvas.width * starsCanvas.height) / 6000);
  for (let i = 0; i < count; i++) {
    stars.push({
      x: Math.random() * starsCanvas.width,
      y: Math.random() * starsCanvas.height,
      r: Math.random() * 1.1,
      a: Math.random() * 0.5 + 0.15,
      twinkle: Math.random() * 0.02 + 0.005,
      phase: Math.random() * Math.PI * 2
    });
  }
}
resizeStars();
window.addEventListener('resize', resizeStars);

function drawStars(t) {
  starsCtx.clearRect(0, 0, starsCanvas.width, starsCanvas.height);
  for (const s of stars) {
    const alpha = s.a + Math.sin(t * s.twinkle + s.phase) * 0.15;
    starsCtx.fillStyle = `rgba(232, 228, 214, ${Math.max(0.05, alpha)})`;
    starsCtx.beginPath();
    starsCtx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
    starsCtx.fill();
  }
}

nodes.forEach((n, i) => { nodeIndex[n.id] = i; });

// Edges: top pole → ring 1, ring 1 → ring 2 (nearest 3), ring 2 ring,
// ring 2 → ring 3 (nearest 3), ring 3 → bottom pole, plus ring1 ring and ring3 ring
const edges = [];

// Top pole to each ring 1 point
for (let i = 0; i < 3; i++) edges.push(['top', `r1-${i}`]);

// Ring 1 around itself
for (let i = 0; i < 3; i++) edges.push([`r1-${i}`, `r1-${(i + 1) % 3}`]);

// Ring 1 → Ring 2: each r1 point connects to the 3 nearest r2 points
function nearestRing2For(r1Angle) {
  return nodes
    .map((n, i) => ({ i, id: n.id, angle: n.angle }))
    .filter(n => n.id.startsWith('r2-'))
    .map(n => {
      let d = Math.abs(n.angle - r1Angle);
      if (d > Math.PI) d = 2 * Math.PI - d;
      return { ...n, d };
    })
    .sort((a, b) => a.d - b.d)
    .slice(0, 3);
}
[0, 1, 2].forEach(i => {
  const r1 = nodes[nodeIndex[`r1-${i}`]];
  const nearest = nearestRing2For(r1.angle);
  nearest.forEach(n => edges.push([`r1-${i}`, n.id]));
});

// Ring 2 around itself
for (let i = 0; i < 8; i++) edges.push([`r2-${i}`, `r2-${(i + 1) % 8}`]);

// Ring 2 → Ring 3: mirror logic
[0, 1, 2].forEach(i => {
  const r3 = nodes[nodeIndex[`r3-${i}`]];
  const nearest = nodes
    .map((n, idx) => ({ i: idx, id: n.id, angle: n.angle }))
    .filter(n => n.id.startsWith('r2-'))
    .map(n => {
      let d = Math.abs(n.angle - r3.angle);
      if (d > Math.PI) d = 2 * Math.PI - d;
      return { ...n, d };
    })
    .sort((a, b) => a.d - b.d)
    .slice(0, 3);
  nearest.forEach(n => edges.push([`r3-${i}`, n.id]));
});

// Ring 3 around itself
for (let i = 0; i < 3; i++) edges.push([`r3-${i}`, `r3-${(i + 1) % 3}`]);

// Bottom pole to each ring 3 point
for (let i = 0; i < 3; i++) edges.push(['bottom', `r3-${i}`]);

wrap.appendChild(renderer.domElement);

// Group that holds the entire mandala (we rotate this for user drag + auto-spin)
const mandala = new THREE.Group();
scene.add(mandala);

// Build node meshes
const nodeMeshes = [];
const nodeGeo = new THREE.SphereGeometry(0.07, 16, 16);
nodes.forEach((node, i) => {
  const mat = new THREE.MeshBasicMaterial({
    color: node.color,
    transparent: true,
    opacity: 0.95
  });
  const mesh = new THREE.Mesh(nodeGeo, mat);
  mesh.userData = { nodeIndex: i };

  // Glow halo
  const haloGeo = new THREE.SphereGeometry(0.16, 16, 16);
  const haloMat = new THREE.MeshBasicMaterial({
    color: node.color,
    transparent: true,
    opacity: 0.18,
    depthWrite: false
  });
  const halo = new THREE.Mesh(haloGeo, haloMat);
  mesh.add(halo);

  mandala.add(mesh);
  nodeMeshes.push({ mesh, halo, node, currentPos: new THREE.Vector3() });
});

// Build edges (lines) — we need to update these per frame as nodes move
const edgeGeometries = [];
const edgeMaterial = new THREE.LineBasicMaterial({
  color: 0xe8e4d6,
  transparent: true,
  opacity: 0.22
});
const edgeMeshes = [];
edges.forEach(([aId, bId]) => {
  const geo = new THREE.BufferGeometry();
  const positions = new Float32Array(6);
  geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
  const line = new THREE.Line(geo, edgeMaterial);
  mandala.add(line);
  edgeMeshes.push({ line, aIndex: nodeIndex[aId], bIndex: nodeIndex[bId] });
});

// =====================================================
// COMPRESSION (sphere → line)
// =====================================================
let compression = 1.0; // 1 = full sphere, 0 = collapsed to axis

function updateNodePositions() {
  nodeMeshes.forEach(({ mesh, node, currentPos }) => {
    const r = node.baseRadius * compression;
    const x = Math.cos(node.angle) * r;
    const z = Math.sin(node.angle) * r;
    const y = node.baseHeight;
    mesh.position.set(x, y, z);
    currentPos.set(x, y, z);
  });

  edgeMeshes.forEach(({ line, aIndex, bIndex }) => {
    const a = nodeMeshes[aIndex].currentPos;
    const b = nodeMeshes[bIndex].currentPos;
    const arr = line.geometry.attributes.position.array;
    arr[0] = a.x; arr[1] = a.y; arr[2] = a.z;
    arr[3] = b.x; arr[4] = b.y; arr[5] = b.z;
    line.geometry.attributes.position.needsUpdate = true;
  });
}
updateNodePositions();

// INTERACTION: rotation + click
// =====================================================
let isDragging = false;
let lastX = 0, lastY = 0;
let rotVelX = 0, rotVelY = 0;
let autoSpin = true;
let lastInteraction = performance.now();

const canvas = renderer.domElement;

canvas.addEventListener('mousedown', (e) => {
  isDragging = true;
  lastX = e.clientX; lastY = e.clientY;
  autoSpin = false;
  lastInteraction = performance.now();
});
window.addEventListener('mousemove', (e) => {
  if (!isDragging) return;
  const dx = e.clientX - lastX;
  const dy = e.clientY - lastY;
  rotVelY = dx * 0.005;
  rotVelX = dy * 0.005;
  mandala.rotation.y += rotVelY;
  mandala.rotation.x += rotVelX;
  lastX = e.clientX; lastY = e.clientY;
  lastInteraction = performance.now();
});
window.addEventListener('mouseup', () => { isDragging = false; });

// Touch support
canvas.addEventListener('touchstart', (e) => {
  if (e.touches.length === 1) {
    isDragging = true;
    lastX = e.touches[0].clientX; lastY = e.touches[0].clientY;
    autoSpin = false;
    lastInteraction = performance.now();
  }
}, { passive: true });
window.addEventListener('touchmove', (e) => {
  if (!isDragging || e.touches.length !== 1) return;
  const dx = e.touches[0].clientX - lastX;
  const dy = e.touches[0].clientY - lastY;
  mandala.rotation.y += dx * 0.005;
  mandala.rotation.x += dy * 0.005;
  lastX = e.touches[0].clientX; lastY = e.touches[0].clientY;
  lastInteraction = performance.now();
}, { passive: true });
window.addEventListener('touchend', () => { isDragging = false; });

// Click detection (raycast) — separate small drag threshold so click vs drag is distinct
let mouseDownPos = null;
canvas.addEventListener('mousedown', (e) => { mouseDownPos = { x: e.clientX, y: e.clientY }; });
canvas.addEventListener('mouseup', (e) => {
  if (!mouseDownPos) return;
  const dx = e.clientX - mouseDownPos.x;
  const dy = e.clientY - mouseDownPos.y;
  if (Math.sqrt(dx * dx + dy * dy) < 4) {
    handleClick(e.clientX, e.clientY);
  }
  mouseDownPos = null;
});
canvas.addEventListener('touchend', (e) => {
  if (e.changedTouches.length !== 1) return;
  // For touch, only treat as click if no significant drag occurred
  const t = e.changedTouches[0];
  if (mouseDownPos) {
    const dx = t.clientX - mouseDownPos.x;
    const dy = t.clientY - mouseDownPos.y;
    if (Math.sqrt(dx * dx + dy * dy) < 8) {
      handleClick(t.clientX, t.clientY);
    }
  } else {
    handleClick(t.clientX, t.clientY);
  }
});

const raycaster = new THREE.Raycaster();
raycaster.params.Mesh.threshold = 0.1;
const mouseVec = new THREE.Vector2();

function handleClick(clientX, clientY) {
  const rect = canvas.getBoundingClientRect();
  mouseVec.x = ((clientX - rect.left) / rect.width) * 2 - 1;
  mouseVec.y = -((clientY - rect.top) / rect.height) * 2 + 1;
  raycaster.setFromCamera(mouseVec, camera);
  // Test against just the node spheres (not halos)
  const meshes = nodeMeshes.map(n => n.mesh);
  const hits = raycaster.intersectObjects(meshes, false);
  if (hits.length > 0) {
    const hit = hits[0].object;
    const idx = hit.userData.nodeIndex;
    addQuestionToPanel(idx);
    // brief pulse effect
    pulseNode(idx);
  }
}

function pulseNode(idx) {
  const { halo } = nodeMeshes[idx];
  halo.userData.pulseStart = performance.now();
}

// SIDE PANEL

const panelBody = document.getElementById('panel-body');
const emptyState = document.getElementById('empty-state');
const clearAllBtn = document.getElementById('clear-all');
const heldNodes = new Map(); // nodeIndex → cardElement

function addQuestionToPanel(idx) {
  if (heldNodes.has(idx)) {
    // Already in panel — flash it instead of duplicating
    const card = heldNodes.get(idx);
    card.style.animation = 'none';
    void card.offsetWidth;
    card.style.animation = 'cardIn 0.4s ease-out';
    return;
  }

  const node = nodes[idx];
  const card = document.createElement('div');
  card.className = 'question-card';

  const colorHex = '#' + node.color.toString(16).padStart(6, '0');
  const isPlaceholder = !node.question || node.question.trim() === '';

  card.innerHTML = `
    <div class="meta">
      <div class="swatch" style="background:${colorHex}; box-shadow:0 0 6px ${colorHex};"></div>
      <span>${node.label}</span>
    </div>
    <div class="text${isPlaceholder ? ' placeholder' : ''}">${
      isPlaceholder ? '(unwritten — what belongs here?)' : node.question
    }</div>
    <button class="close-btn" aria-label="Remove question">×</button>
  `;
  card.querySelector('.close-btn').addEventListener('click', () => {
    card.remove();
    heldNodes.delete(idx);
    refreshPanelState();
  });

  if (emptyState.parentNode) emptyState.remove();
  panelBody.appendChild(card);
  heldNodes.set(idx, card);
  refreshPanelState();
}

function refreshPanelState() {
  if (heldNodes.size === 0) {
    if (!document.getElementById('empty-state')) {
      panelBody.appendChild(emptyState);
    }
    clearAllBtn.classList.remove('visible');
  } else {
    clearAllBtn.classList.add('visible');
  }
}

clearAllBtn.addEventListener('click', () => {
  heldNodes.forEach(card => card.remove());
  heldNodes.clear();
  refreshPanelState();
});

// ANIMATION LOOP
// =====================================================
function animate(t) {
  requestAnimationFrame(animate);

  // Resume gentle auto-spin if user has been idle
  if (!isDragging && (performance.now() - lastInteraction) > 2500) {
    autoSpin = true;
  }
  if (autoSpin && !isDragging) {
    mandala.rotation.y += 0.0015;
    mandala.rotation.x *= 0.998; // gently return tilt
  }

  // Halo pulses
  nodeMeshes.forEach(({ halo, mesh }) => {
    if (halo.userData.pulseStart) {
      const elapsed = performance.now() - halo.userData.pulseStart;
      if (elapsed < 800) {
        const phase = elapsed / 800;
        halo.scale.setScalar(1 + (1 - phase) * 1.4);
        halo.material.opacity = 0.18 + (1 - phase) * 0.5;
      } else {
        halo.scale.setScalar(1);
        halo.material.opacity = 0.18;
        halo.userData.pulseStart = null;
      }
    }
  });

  drawStars(t);
  renderer.render(scene, camera);
}
requestAnimationFrame(animate);

let focusedNodeIndex = -1;
window.addEventListener('keydown', (e) => {
  if (e.key === 'Tab' && !e.shiftKey) {
    if (document.activeElement === document.body || document.activeElement === canvas) {
      e.preventDefault();
      focusedNodeIndex = (focusedNodeIndex + 1) % nodes.length;
      pulseNode(focusedNodeIndex);
    }
  } else if (e.key === 'Enter' && focusedNodeIndex >= 0) {
    addQuestionToPanel(focusedNodeIndex);
  }
});
